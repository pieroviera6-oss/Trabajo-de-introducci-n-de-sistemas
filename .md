Función ValidarCosecha(entrada,estado Por Referencia,kg Por Referencia)
	Definir i, separadores, digitos Como Entero
	Definir letra, numeroTexto Como Cadena
	Definir correcto Como Lógico
	estado <- 'INVALIDO'
	kg <- 0
	correcto <- Verdadero
	separadores <- 0
	digitos <- 0
	numeroTexto <- ''
	Si Longitud(entrada)=0 Entonces
		estado <- 'INCOMPLETO'
	SiNo
		Para i<-0 Hasta Longitud(entrada)-1 Hacer
			letra <- SubCadena(entrada,i,i)
			Si letra='.' O letra=',' Entonces
				separadores <- separadores+1
				numeroTexto <- Concatenar(numeroTexto,'.')
			SiNo
				Si letra='0' O letra='1' O letra='2' O letra='3' O letra='4' O letra='5' O letra='6' O letra='7' O letra='8' O letra='9' Entonces
					digitos <- digitos+1
					numeroTexto <- Concatenar(numeroTexto,letra)
				SiNo
					correcto <- Falso
				FinSi
			FinSi
		FinPara
		Si separadores>1 O digitos=0 Entonces
			correcto <- Falso
		FinSi
		Si correcto Entonces
			kg <- ConvertirANumero(numeroTexto)
			Si kg<=0 Entonces
				estado <- 'INVALIDO'
			SiNo
				Si kg>1000 Entonces
					estado <- 'REVISAR'
				SiNo
					estado <- 'VALIDO'
				FinSi
			FinSi
		FinSi
	FinSi
FinFunción
